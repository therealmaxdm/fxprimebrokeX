                      <span className="font-bold text-white">Florence</span>

                      <span className="text-slate-300">Bank</span>
                      <span className="font-bold text-white">FairMoney Microfinance Bank LTD</span>
                    </div>
                    <div className="flex justify-between bg-indigo-900/40 p-3 rounded-2xl border border-indigo-800/60">
                      <span className="text-slate-300">Amount</span>
                      <span className="font-black text-emerald-400 text-sm">₦{Number(upgradeAmount).toLocaleString()}</span>

                  </div>

                  <div className="pt-2">
                    <label className="text-xs font-semibold text-slate-300 block mb-1">Upload Payment Receipt</label>

                  CONFIRM MY UPGRADE
                </button>
              </div>
            )}

            {upgradeStep === 'loading' && (
              <div className="bg-indigo-950/40 border border-indigo-900/60 rounded-3xl p-8 text-center space-y-4 shadow-xl">
                <div className="w-12 h-12 rounded-full border-4 border-amber-500 border-t-transparent animate-spin mx-auto"></div>

                <div className="text-xs text-slate-300">Please wait ({upgradeCountdown}s)...</div>

                  <X className="w-5 h-5 bg-red-500/20 rounded-full p-1" />
                  <span className="font-bold text-sm">Payment Status</span>
                </div>

                  Payment not received ❌ Please add funds again to continue.
                </p>
                <button onClick={() => setUpgradeStep('form')} className="w-full bg-gradient-to-r from-orange-500 to-amber-500 hover:opacity-90 text-slate-950 font-black py-3.5 rounded-2xl shadow-xl transition">
                  Go Back to Upgrade Details
                </button>
              </div>
            )}
          </div>
        )}

        {/* ================= 7. PROMOTION CENTER ================= */}
        {currentScreen === 'promotion' && (
          <div className="space-y-5 animate-fadeIn">
            <div>
              <h2 className="text-xl font-bold">Promotion Center</h2>

        {(currentScreen === 'tiktokPromo' || currentScreen === 'youtubePromo' || currentScreen === 'telegramPromo' || currentScreen === 'otherPromo') && (

            <div className="flex items-center justify-between">
              <h2 className="text-xl font-bold capitalize">{currentScreen.replace('Promo', '')} Promotion</h2>

            <div className="bg-indigo-950/40 border border-indigo-900/60 rounded-3xl p-4 space-y-4 shadow-xl">

                    <option value="subscribers">Subscribers: 40 / ₦1,000</option>

                  <label className="text-xs font-semibold text-slate-300 block mb-1">What do you want promoted?</label>

              </div>

              <div>
                <label className="text-xs font-semibold text-slate-300 block mb-1">Amount (₦) Minimum ₦1,000</label>

              </div>

              <button onClick={() => {
                if(promoUrl && promoAmount >= 1000) {
                  setUpgradeAmount(promoAmount);
                  navigateTo('upgrade');
                } else {
                  alert('Please enter a valid URL and minimum amount of ₦1,000');
                }
              }} className="w-full bg-emerald-500 hover:bg-emerald-600 text-slate-950 font-black py-3.5 rounded-2xl shadow-xl transition">
                Proceed to Payment & Upgrade
              </button>
            </div>
          </div>
        )}

        {/* ================= 8. PROFILE MENU ================= */}
        {currentScreen === 'profile' && (

            <div className="flex items-center space-x-3 bg-indigo-950/40 p-4 rounded-3xl border border-indigo-900/60 shadow-xl">

                <img src={profilePic} alt="Profile" className="w-14 h-14 rounded-2xl object-cover border border-emerald-400" />
              ) : (
                <div className="w-14 h-14 rounded-2xl bg-gradient-to-tr from-emerald-400 to-teal-400 flex items-center text-slate-950 font-black justify-center text-xl shadow">

            <div className="space-y-2">
              {[
                { title: 'Get App', desc: 'Install Cash Pay Wallet on your device', icon: Smartphone, action: () => alert('App download triggered!') },
                { title: 'Profile Information', desc: 'Your registration details & password', icon: User, action: () => alert('Profile details view') },
                { title: 'About Cash Pay Wallet', desc: 'How the app works & FAQ', icon: Shield, action: () => alert('Cash Pay Wallet is a modern financial platform...') },
                { title: 'Add Bank', desc: 'Manage your withdrawal bank accounts', icon: Wallet, action: () => navigateTo('addBank') },
                { title: 'Log out', desc: 'Sign out of your Cash Pay Wallet account', icon: Lock, action: () => { setIsLoggedIn(false); navigateTo('home'); }, danger: true },

                const IconComponent = item.icon;
                return (
                  <div key={idx} onClick={item.action} className={`bg-indigo-950/40 hover:bg-indigo-950/80 border border-indigo-900/60 hover:border-indigo-700 p-3.5 rounded-2xl flex items-center justify-between cursor-pointer transition shadow ${item.danger ? 'text-red-400' : 'text-white'}`}>
                    <div className="flex items-center space-x-3">
                      <div className={`p-2.5 rounded-xl ${item.danger ? 'bg-red-500/20 text-red-400' : 'bg-indigo-900/60 text-emerald-400 shadow'}`}>
                        <IconComponent className="w-4 h-4" />
                      </div>
                      <div>
                        <h4 className="font-bold text-sm">{item.title}</h4>
                        <p className="text-xs text-slate-300">{item.desc}</p>
                      </div>
                    </div>
                    <ChevronRight className="w-4 h-4 text-slate-400" />
                  </div>
                );
              })}
            </div>
          </div>
        )}

      </main>

      {/* Bottom Navigation Bar */}
      {isLoggedIn && currentScreen !== 'home' && currentScreen !== 'register' && (
        <nav className="fixed bottom-0 left-0 right-0 bg-slate-950/90 backdrop-blur-md border-t border-indigo-900/60 py-3 px-6 flex justify-around items-center z-40 shadow-2xl">
          <button onClick={() => navigateTo('dashboard')} className={`flex flex-col items-center space-y-1 transition ${currentScreen === 'dashboard' ? 'text-emerald-400' : 'text-slate-400 hover:text-slate-200'}`}>
            <Home className="w-5 h-5" />
            <span className="text-[10px] font-bold">Home</span>
          </button>
          <button onClick={() => navigateTo('promotion')} className={`flex flex-col items-center space-y-1 transition ${currentScreen === 'promotion' || currentScreen.includes('Promo') ? 'text-emerald-400' : 'text-slate-400 hover:text-slate-200'}`}>
            <Layers className="w-5 h-5" />
            <span className="text-[10px] font-bold">Ads</span>
          </button>
          <button onClick={() => navigateTo('profile')} className={`flex flex-col items-center space-y-1 transition ${currentScreen === 'profile' ? 'text-emerald-400' : 'text-slate-400 hover:text-slate-200'}`}>
            <User className="w-5 h-5" />
            <span className="text-[10px] font-bold">Profile</span>
          </button>
        </nav>
      )}
    </div>
  );
